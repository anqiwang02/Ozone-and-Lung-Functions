test_stat <- list() #every CpG site has all possible 131072 test statistics
ob_stat <- matrix(nrow=row, ncol=1) #every CpG only has one observed statistics; so total of row=283 number of observed statistics for secondary gene list
unadjust_p <-matrix(nrow=row, ncol=1) #every CpG site only has one p value

for(h in 1:row){ #h represents the number of cpg column values
   test_stat[[h]] <-abs(w0.true%*%as.matrix(final[,h+4])/17)  #store all possible test statistics of each column inside a list of subjects
  ob_stat[h,1]<-abs(t(final$exp)%*%as.matrix(final[,h+4])/17) #store observed test statistics of each column
  unadjust_p[h,1]<-colSums((1*(sapply(seq_along(ob_stat[h,1]), function(x, y, i) y[i] <= x[,i], y = ob_stat[h,1] ,x = test_stat[[h]])))/2^17) #p-value, comparing one observed statistics (y) with all possible test statistics x for each CpG site
}
